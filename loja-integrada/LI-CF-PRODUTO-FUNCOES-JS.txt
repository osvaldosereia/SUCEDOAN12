/* CANECAFÁCIL · PRODUTO FUNÇÕES v1
   Cole como JavaScript no rodapé. Não use <script>. */
(function(){
'use strict';
if(window.CF_PRODUCT_FLOW_V1)return;window.CF_PRODUCT_FLOW_V1=1;
if(!document.body||!document.body.classList.contains('pagina-produto'))return;

var FIREBASE='https://cedar-chemist-310801-default-rtdb.firebaseio.com';
var required=false,box=null,buy=null;
function q(s,r){return(r||document).querySelector(s)}
function qa(s,r){return Array.prototype.slice.call((r||document).querySelectorAll(s))}
function text(n){return n?String(n.textContent||'').replace(/\s+/g,' ').trim():''}
function personalizer(){return q('.cf-personalizer-box')}
function iframe(){var b=personalizer();return b?q('iframe[src*="/loja-integrada/personalizar/"]',b):null}
function actions(){return q('.acoes-produto')||q('.principal')}
function quantity(){var a=actions();return a&&(q('.qtde-adicionar-carrinho',a)||q('.quantidade',a)||q('[class*="quantidade"]',a))}
function buyButton(){var a=actions();return a&&(q('.botao-comprar',a)||q('.comprar .botao',a)||q('.comprar button',a)||q('.comprar a',a))}
function modelKey(){var f=iframe();if(!f)return'';try{return new URL(f.src,location.href).searchParams.get('model')||''}catch(e){return''}}

function movePersonalizer(){
  box=personalizer();var a=actions();if(!box||!a)return false;
  var qty=quantity(),comprar=q('.comprar',a),anchor=qty||comprar;
  if(anchor&&box.parentNode!==a){a.insertBefore(box,anchor)}
  else if(anchor&&box.nextElementSibling!==anchor){a.insertBefore(box,anchor)}
  box.setAttribute('data-cf-product-position','before-buy');
  return true;
}

function note(){
  var a=actions();if(!a)return;
  var n=q('.cf-required-buy-note',a);
  if(!required){if(n&&n.parentNode)n.parentNode.removeChild(n);return}
  if(!n){n=document.createElement('div');n.className='cf-required-buy-note';n.textContent='Personalização obrigatória neste modelo. Gere e aprove sua arte acima para continuar a compra.';var qty=quantity(),comprar=q('.comprar',a);a.insertBefore(n,qty||comprar||null)}
}

function lockBuy(){
  buy=buyButton();if(!buy)return;
  if(!required)return;
  buy.setAttribute('aria-disabled','true');
  buy.setAttribute('data-cf-required-lock','1');
  if('disabled' in buy)buy.disabled=true;
  if(!buy.dataset.cfOriginalLabel)buy.dataset.cfOriginalLabel=text(buy)||'Comprar';
  buy.textContent='Personalize para comprar';
}

function applyRequired(p){
  var cfg=p&&p.personalizacao&&typeof p.personalizacao==='object'?p.personalizacao:{};
  required=cfg.ativa===true&&cfg.obrigatoria===true;
  document.body.classList.toggle('cf-required-personalization',required);
  box=personalizer();if(box)box.setAttribute('data-cf-personalization-required',required?'1':'0');
  note();lockBuy();contextMobile();
}

async function loadConfig(){
  var key=modelKey();if(!key)return;
  try{var r=await fetch(FIREBASE+'/produtos/'+encodeURIComponent(key)+'.json?_='+Date.now(),{cache:'no-store'});if(!r.ok)return;var p=await r.json();if(p)applyRequired(p)}catch(e){}
}

function scrollPersonalizer(){
  var b=personalizer();if(!b)return;
  b.scrollIntoView({behavior:'smooth',block:'center'});
  setTimeout(function(){var f=iframe();try{f&&f.contentWindow&&f.contentWindow.focus()}catch(e){}},450);
}

function guardClick(e){
  if(!required)return;
  var t=e.target&&e.target.closest?e.target.closest('.botao-comprar,.comprar button,.comprar a'):null;
  if(!t)return;
  e.preventDefault();e.stopPropagation();e.stopImmediatePropagation();scrollPersonalizer();
}
function guardSubmit(e){if(!required)return;var a=actions();if(a&&a.contains(e.target)){e.preventDefault();e.stopPropagation();scrollPersonalizer()}}

document.addEventListener('click',guardClick,true);
document.addEventListener('submit',guardSubmit,true);

function contextMobile(){
  var create=q('#cfMobileNav .cf-mobile-nav-create');if(!create)return;
  var label=q('span:last-child',create);if(label)label.textContent='Personalizar';
  create.removeAttribute('href');create.setAttribute('href','#personalizar');
  if(create.dataset.cfProductBound==='1')return;create.dataset.cfProductBound='1';
  create.addEventListener('click',function(e){e.preventDefault();scrollPersonalizer()});
}

function markGallery(){
  qa('.produto-thumbs li').forEach(function(li,i){li.setAttribute('data-cf-photo-index',String(i+1));var img=q('img',li);if(img){img.loading='lazy';img.setAttribute('data-cf-photo-index',String(i+1))}});
}

function cleanLabels(){
  var b=personalizer();if(!b)return;var p=b.previousElementSibling;
  if(p&&/^personalize (esta|sua) caneca:?$/i.test(text(p))&&p.parentNode)p.parentNode.removeChild(p);
}

function refresh(){
  movePersonalizer();cleanLabels();markGallery();contextMobile();
  if(modelKey()&&!document.body.dataset.cfConfigLoaded){document.body.dataset.cfConfigLoaded='1';loadConfig()}
  if(required){note();lockBuy()}
}

refresh();setTimeout(refresh,250);setTimeout(refresh,800);setTimeout(refresh,1800);
var root=q('.produto')||document.body;var observer=new MutationObserver(function(){refresh()});observer.observe(root,{childList:true,subtree:true});
setTimeout(function(){observer.disconnect()},8000);
})();
